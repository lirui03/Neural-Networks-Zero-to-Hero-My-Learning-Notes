## 1.3 计算图可视化


```
#从 root 开始，深度优先遍历整个图。root在此处表示运算的终点
def trace(root):
    nodes, edges = set(), set()
    def build(v):
        if v not in nodes:
            nodes.add(v)
            for child in v._prev:
                edges.add((child, v))
                build(child)
    build(root)
    return nodes, edges

def draw_dot(root, format='svg', rankdir='LR'):
    """
    format: png | svg | ...
    rankdir: TB (top to bottom graph) | LR (left to right)
    """
    assert rankdir in ['LR', 'TB']
    nodes, edges = trace(root)
    dot = Digraph(format=format, graph_attr={'rankdir': rankdir}) #, node_attr={'rankdir': 'TB'})
    
    for n in nodes:
        dot.node(name=str(id(n)), label = "{ data %.4f | grad %.4f }" % (n.data, n.grad), shape='record')
        if n._op:
            dot.node(name=str(id(n)) + n._op, label=n._op)
            dot.edge(str(id(n)) + n._op, str(id(n)))
    
    for n1, n2 in edges:
        dot.edge(str(id(n1)), str(id(n2)) + n2._op)
    
    return dot
```
trace 负责收集计算图的所有节点和边，draw_dot 负责把它们渲染成 Graphviz 图，从而让数学表达式和梯度传播过程"可视化"。

## 1.4 构建Value类


```
class Value:
    
    def __init__(self, data, _children=(), _op='', label=''):
        self.data = data
        self.grad=0.0
        self._backward=lambda: None
        self._prev = set(_children)
        self._op = _op
        self.label = label
    
    def __repr__(self) -> str:
        return f"Value(data={self.data})"
    
    def __add__(self, other):
        other = other if isinstance(other, Value) else Value(other)
        out = Value(self.data + other.data, (self, other), '+')
        def _backward():
            self.grad += 1*out.grad
            other.grad += 1*out.grad
        out._backward = _backward
        return out
    
    def __neg__(self):
        return self*(-1)
        
    def __sub__(self, other):
        return self+(-other)
    
    def __radd__(self, other):
        return self+other
    
    def __mul__(self, other):
        other = other if isinstance(other, Value) else Value(other)
        out = Value(self.data * other.data, (self, other), '*')
        def _backward():
            self.grad += other.data*out.grad
            other.grad += self.data*out.grad
        out._backward = _backward
        return out
    
    def __pow__(self, other):
        assert isinstance(other,(int, float)),"仅支持int/float类型"
        out = Value(self.data**other, (self,),f"**{other}")
        def _backward():
            self.grad += other*self.data**(other-1)*out.grad
        out._backward = _backward
        return out
        
    def __rmul__(self, other):
        return self*other
        
    def __truediv__(self, other):
        return self*other**-1
    
    def tanh(self):
        x = self.data
        t = (math.exp(2*x)-1)/(math.exp(2*x)+1)
        out = Value(t, (self, ))
        def _backward():
            self.grad += (1-t**2) * out.grad
        out._backward = _backward
        return out
    
    def exp(self):
        x = self.data
        out = Value(math.exp(x), (self,), 'exp')
        def _backward():
            self.grad += out.data*out.grad#math.exp(x)*out.grad
        out._backward = _backward
        return out
        
    def backward(self):
        self.grad = 1.0
        topo = []#拓扑序列
        visited = set()#已访问节点的集合
        def build_topo(v):
            if v not in visited:
                visited.add(v)
                for child in v._prev:
                    build_topo(child)
                topo.append(v)
        build_topo(self)
        for node in reversed(topo):
            node._backward()
```

python中，这种前后被双下划线包裹的函数被称为魔术方法，它们不需要被用户显示调用，而是在特定场景下自动触发。例如代码中的\_\_add\_\_会在执行a+b时自动调用。同理，当我们创建一个Value对象的时候，会自动调用init函数。

- data表示具体数值。

- _children表示子节点。例如a+b=c，这里的a和b就是c的子节点；再例如，d*e+f=g，这里的d、e、f均是g的子节点。之所以用元组表示，是因为python中元组不可改变，作为参数可以保证安全。可能有人会想到用列表，使用列表会出现一系列问题。如下图代码所示：

  ```
  def add_item(item, lst=[]):
      lst.append(item)
      return lst
  
  print(add_item(1))   # [1]
  print(add_item(2))   # [1, 2]   ← 你以为是 [2]，其实是 [1, 2]
  print(add_item(3))   # [1, 2, 3]
  ```

  Python 函数的默认参数是在定义时求值的，而不是每次调用时求值。如果使用_children=[]，那么定义这个函数的时候，列表对象就会存在，被存进f.\_\_defaults\_\_中。每次调用时，实际都是在操纵同一个列表。因此不使用列表，而是选择使用元组。

- _op表示运算符，子节点通过何种运算得到的现节点。

- label表示标签，可视化的时候可以根据标签理解数学表达式。

- grad表示梯度。

- _backward这里存储的是一个函数对象。

当出现a+b=c时，会自动调用\_\_add\_\_()，因此这个表达式可以改写成c=a.\_\_add\_\_(b)，函数里面的self就是a，other就是b。

最关键部分为：

```
def _backward():
    self.grad += 1*out.grad
    other.grad += 1*out.grad
out._backward = _backward
```

首先定义了一个函数_backward()，该函数用来计算梯度。out=(self)op操作(other)，已经知道out的梯度下如何求self和other的梯度？

**对于加法，op操作已经知道。out=(self) + (other)，结果显而易见，self和other的梯度均为1*out.grad。**

为什么不直接赋值，反而要用+=？

**事实上，用=可能会满足一些情况，但是另外一些特殊情况则无法满足。例如a+a=b，使用赋值，由于后一个会覆盖前一个，导致梯度始终为1！但其实梯度为2，结果显然错误。使用+=则完美解决了这个问题。**

out._backward = _backward将函数对象赋值给out的\_backward属性，千万不要加上括号！使用括号后右边则代表调用，是将函数的返回值赋给\_backward属性，而非函数对象。

这样每一个节点的\_backward属性都会绑定一个函数，a.\_backward()即可调用节点a的反向传播方法，但是叶节点除外。因为这里是将函数绑定到运算结果节点的属性上，所以叶子节点的该属性永远为空函数lambda：None，叶子节点的梯度是由其运算结果节点的\_backward()计算的。因此a.\_backward()计算的其实是它子节点的梯度。【子节点指的是前驱节点】

我们已经可以计算a+b了，但是a+2怎么计算？

class Value:
    
    def __init__(self, data, _children=(), _op='', label=''):
        self.data = data
        self.grad=0.0
        self._backward=lambda: None
        self._prev = set(_children)

        out = Value(self.data + other.data, (self, other), '+')
        def _backward():
            self.grad += 1*out.grad
            other.grad += 1*out.grad
        out._backward = _backward
        return out
    
    def __neg__(self):
        return self*(-1)
        
    def __sub__(self, other):
        return self+(-other)
    
    def __radd__(self, other):
        return self+other
    
    def __mul__(self, other):
        other = other if isinstance(other, Value) else Value(other)
        out = Value(self.data * other.data, (self, other), '*')
        def _backward():
            self.grad += other.data*out.grad
            other.grad += self.data*out.grad
        out._backward = _backward
        return out
    
    def __pow__(self, other):
        assert isinstance(other,(int, float)),"仅支持int/float类型"
        out = Value(self.data**other, (self,),f"**{other}")
        def _backward():
            self.grad += other*self.data**(other-1)*out.grad
        out._backward = _backward
        return out
        
    def __rmul__(self, other):
        return self*other
        
    def __truediv__(self, other):
        return self*other**-1
    
    def tanh(self):
        x = self.data
        t = (math.exp(2*x)-1)/(math.exp(2*x)+1)
        out = Value(t, (self, ))
        def _backward():
            self.grad += (1-t**2) * out.grad
        out._backward = _backward
        return out
    
    def exp(self):
        x = self.data
        out = Value(math.exp(x), (self,), 'exp')
        def _backward():
            self.grad += out.data*out.grad#math.exp(x)*out.grad
        out._backward = _backward
        return out
        
    def backward(self):
        self.grad = 1.0
        topo = []#拓扑序列
        visited = set()#已访问节点的集合
        def build_topo(v):
            if v not in visited:
                visited.add(v)
                for child in v._prev:
                    build_topo(child)
                topo.append(v)
        build_topo(self)
        for node in reversed(topo):
            node._backward()

class Neuron:
    def __init__(self, nin):#nin表示接受几个输入，例如Neuron(3)表示接受三个输入
        self.w = [Value(random.uniform(-1,1)) for _ in range(nin)]
        self.b = Value(random.uniform(-1,1))

    def __call__(self, x):
        act = sum((wi*xi for wi,xi in zip(self.w, x)),self.b)
        out = act.tanh()
        return out
    
    def parameters(self):
        return self.w+[self.b]

class Layer:
    def __init__(self, nin, nout):
        self.neurons = [Neuron(nin) for _ in range(nout)]

    def __call__(self, x):
        outs = [n(x) for n in self.neurons]
        return outs[0] if len(outs) == 1 else outs
    
    def parameters(self):
        params = []
        for neuron in self.neurons:
            ps = neuron.parameters()
            params.extend(ps)
        return params

class MLP:
    #nin表示每个神经元的权重，nout表示一层有多少个神经元，nouts表示整个网络每一层的神经元个数
    #例如，[3, 5, 3]表示第一层有三个神经元，第二层有五个，最后一层也有三个
    def __init__(self, nin, nouts):
        #sz表示尺寸，例如首个输入有2个特征，nouts=[3,4]，那么会得到列表[2,3,4] 
        sz = [nin] + nouts
        #len(nouts)表示一共有几层，用i选定某一层。MLP为全连接，第一层有nout个神经元，会产生nout个输出，这些会组成一个列表，
        #输入下一层的每一个神经元。由于Layer的输入nin必定和上一个的输出相等，所以满足
        #self.layers是一个列表，每一项表示一个layer对象。例如self.layers=[Layer(2,3),layer(3,1)]
        self.layers = [Layer(sz[i],sz[i+1]) for i in range(len(nouts))]
        
    def __call__(self, x):
        for layer in self.layers:
            x = layer(x)
        return x
    
    def parameters(self):
        return [p for layer in self.layers for p in layer.parameters()]
